#include <iostream>
#include <algorithm>
#include <clocale>
using namespace std;

template <typename T>
class Vector {
private:
    T* data;
    int sz;
    int cap;

public:
    // Конструктор
    Vector() {
        data = nullptr;
        sz = 0;
        cap = 0;
    }

    // 1. Конструктор копирования
    Vector(const Vector& other) {
        sz = other.sz;
        cap = other.cap;
        data = new T[cap];

        for (int i = 0; i < sz; i++) {
            data[i] = other.data[i];
        }
    }

    // 2. Оператор копирующего присваивания
    Vector& operator=(const Vector& other) {
        if (this == &other) {
            return *this;
        }

        delete[] data;

        sz = other.sz;
        cap = other.cap;
        data = new T[cap];

        for (int i = 0; i < sz; i++) {
            data[i] = other.data[i];
        }

        return *this;
    }

    // 3. Конструктор перемещения
    Vector(Vector&& other) noexcept {
        data = other.data;
        sz = other.sz;
        cap = other.cap;

        other.data = nullptr;
        other.sz = 0;
        other.cap = 0;
    }

    // 4. Оператор перемещающего присваивания
    Vector& operator=(Vector&& other) noexcept {
        if (this == &other) {
            return *this;
        }

        delete[] data;

        data = other.data;
        sz = other.sz;
        cap = other.cap;

        other.data = nullptr;
        other.sz = 0;
        other.cap = 0;

        return *this;
    }

    // 5. Деструктор
    ~Vector() {
        delete[] data;
    }

    // Добавление элемента в конец
    void push_back(const T& value) {
        if (sz == cap) {
            int newCap = (cap == 0) ? 1 : cap * 2;
            T* newData = new T[newCap];

            for (int i = 0; i < sz; i++) {
                newData[i] = data[i];
            }

            delete[] data;
            data = newData;
            cap = newCap;
        }

        data[sz] = value;
        sz++;
    }

    // Итераторы
    T* begin() {
        return data;
    }

    T* end() {
        return data + sz;
    }

    const T* begin() const {
        return data;
    }

    const T* end() const {
        return data + sz;
    }

    // Циклический сдвиг влево
    void rotate(int k) {
        if (sz == 0) {
            return;
        }

        k = k % sz;

        if (k < 0) {
            k += sz;
        }

        std::rotate(begin(), begin() + k, end());
    }

    // Вывод вектора
    void print() const {
        for (int i = 0; i < sz; i++) {
            cout << data[i] << " ";
        }
        cout << endl;
    }
};

int main() {
    setlocale(LC_ALL, "Russian");

    Vector<int> v;

    v.push_back(1);
    v.push_back(2);
    v.push_back(3);
    v.push_back(4);
    v.push_back(5);

    cout << "Исходный вектор: ";
    v.print();

    v.rotate(2);

    cout << "После rotate(2): ";
    v.print();

    v.rotate(1);

    cout << "После rotate(1): ";
    v.print();

    v.rotate(-2);

    cout << "После rotate(-2): ";
    v.print();

    return 0;
}

Контрольные вопросы:
1. Почему динамический массив называется «динамическим»?
Потому что его размер и ёмкость могут изменяться во время выполнения программы.
2. Что такое ёмкость (capacity) и чем она отличается от размера (size)?
size — количество элементов в массиве. capacity — количество элементов, для которых уже выделена память.
3. Почему при расширении массива ёмкость удваивается, а не увеличивается на фиксированное число?
Чтобы реже выделять новую память и копировать элементы. Удвоение позволяет добавлять элементы со средней амортизированной сложностью \(O(1)\).
4. Какова амортизированная сложность операции push_back? Почему?
\(O(1)\) — амортизированная сложность. Обычно элемент добавляется за постоянное время, а дорогое расширение массива происходит редко.
5. Что такое «правило трёх»? Когда оно применяется?
Правило трёх означает, что если классу требуется собственный деструктор, конструктор копирования или оператор присваивания, то обычно нужно реализовать все три. Оно применяется при ручном управлении ресурсами, например динамической памятью.
6. Что такое «правило пяти»? Чем оно отличается от «правила трёх»?
Правило пяти включает три специальных метода из правила трёх и ещё два: конструктор перемещения и оператор присваивания с перемещением. Они позволяют эффективно передавать ресурсы между объектами.
7. В чём разница между конструктором копирования и оператором присваивания?
Конструктор копирования создаёт новый объект на основе существующего. Оператор присваивания копирует данные в уже созданный объект.
8. Зачем нужна проверка if (this != &other) в операторе присваивания?
Она проверяет, не присваивается ли объект самому себе. Это помогает избежать ошибок при освобождении и повторном копировании памяти.
9. Что такое итератор? Как он связан с указателем?
Итератор — это объект для последовательного доступа к элементам контейнера. Он похож на указатель, поскольку позволяет обращаться к элементу и переходить к следующему.
10. Зачем нужны методы begin() и end()?
begin() возвращает итератор на первый элемент контейнера, а end() — итератор на позицию сразу после последнего элемента.
11. Как реализовать range-based for для своего контейнера?
Нужно реализовать методы begin() и end(), возвращающие итераторы, которые поддерживают необходимые операции: сравнение, разыменование и переход к следующему элементу.
12. Что такое «глубокое копирование»? Чем оно отличается от «поверхностного»?
Глубокое копирование создаёт отдельную копию данных в памяти. Поверхностное копирование копирует значения полей, включая адрес указателя, поэтому объекты могут ссылаться на одну область памяти.
13. Что такое семантика перемещения и зачем она нужна?
Семантика перемещения позволяет передавать ресурсы от одного объекта другому вместо копирования данных. Это повышает эффективность работы с динамической памятью и большими объектами.
14. Почему конструктор перемещения помечается noexcept?
noexcept указывает, что конструктор перемещения не должен выбрасывать исключения. Это позволяет некоторым контейнерам, например vector, эффективнее перемещать элементы при расширении памяти.
15. Как правильно освобождать память в деструкторе?
Если память была выделена через new[], её нужно освободить через delete[]. Если использовался new, нужно использовать delete. Каждую выделенную область памяти следует освободить один раз.
